/*
 * jlk_smart_test.c - 尝试通过不同CE_N值访问Smart寄存器0x0004
 * DSP代码中 reset_1263() 写 Smart_Reg(0x0004)=0, 然后=1
 * Smart_Reg使用不同的地址空间(0xB0000000 vs 0xA0000000)
 * 可能对应不同的CE_N值
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <string.h>
#include <unistd.h>
#include "xdma_hal.h"
#include "jlk1263_linux.h"

int main(int argc, char **argv)
{
    xdma_dev_t dev;
    uint16_t val;
    int rc, i;
    uint8_t ce_values[] = {0x7, 0xb, 0xd, 0xe, 0xf, 0x3, 0x0};

    memset(&dev, 0, sizeof(dev));
    rc = xdma_open_sysfs(&dev, "/sys/bus/pci/devices/0000:87:00.0/resource0");
    if (rc) {
        fprintf(stderr, "Failed to open device: %d\n", rc);
        return 1;
    }
    printf("[OK] device opened\n");

    /* 1. 桥复位 */
    xdma_bridge_reset(&dev, 0);
    usleep(10000);
    xdma_bridge_reset(&dev, 1);
    usleep(50000);

    /* 2. 读取复位前REG(0x0B) */
    printf("\n=== Before reset ===\n");
    jlk_reg_read(&dev, 0x0B, &val);
    printf("REG(0x0B) = 0x%04X (bit15=%d)\n", val, (val>>15)&1);

    /* 3. JLK1263 reset (Linux version - only P_JLK_RESET) */
    fpga_reg_write(&dev, 0x4000, 0x1);
    usleep(100000);
    fpga_reg_write(&dev, 0x4000, 0x0);
    usleep(100000);
    fpga_reg_write(&dev, 0x4000, 0x1);
    usleep(100000);
    printf("[OK] JLK1263 reset done\n");

    /* 4. 读取reset后REG(0x0B) */
    jlk_reg_read(&dev, 0x0B, &val);
    printf("REG(0x0B) = 0x%04X (bit15=%d)\n", val, (val>>15)&1);

    /* 5. 尝试通过不同CE_N值写地址0x0004 */
    printf("\n=== Trying different CE_N values for Smart_Reg(0x0004) ===\n");
    for (i = 0; i < (int)(sizeof(ce_values)/sizeof(ce_values[0])); i++) {
        uint8_t ce = ce_values[i];
        uint32_t rd_val;

        /* 保存当前CE */
        uint8_t old_ce = dev.cur_ce;

        /* 设置CE */
        emif_set_ce(&dev, ce);

        /* 写0x0004 = 0 */
        emif_raw_write(&dev, 0x0004, 0x0);
        usleep(10000);

        /* 写0x0004 = 1 */
        emif_raw_write(&dev, 0x0004, 0x1);
        usleep(10000);

        /* 读REG(0x0B)检查是否变化 */
        emif_set_ce(&dev, 0xb);
        jlk_reg_read(&dev, 0x0B, &val);

        printf("  CE_N=0x%X: REG(0x0B)=0x%04X (bit15=%d) %s\n",
               ce, val, (val>>15)&1,
               (val & 0x8000) ? "" : "<<< CHANGED!");

        /* 恢复CE */
        emif_set_ce(&dev, old_ce);
    }

    /* 6. 也尝试写JLK1263的REG(0x08)和REG(0x09) */
    printf("\n=== Trying JLK1263 REG(0x08) and REG(0x09) ===\n");
    printf("  Before: REG(0x08)=0x%04X, REG(0x09)=0x%04X\n",
           (jlk_reg_read(&dev, 0x08, &val), val),
           (jlk_reg_read(&dev, 0x09, &val), val));

    /* 尝试写REG(0x08) */
    jlk_reg_write(&dev, 0x08, 0x0001);
    usleep(10000);
    jlk_reg_read(&dev, 0x08, &val);
    printf("  After write 0x0001: REG(0x08)=0x%04X\n", val);
    jlk_reg_read(&dev, 0x0B, &val);
    printf("  REG(0x0B)=0x%04X (bit15=%d) %s\n", val, (val>>15)&1,
           (val & 0x8000) ? "" : "<<< CHANGED!");

    /* 恢复 */
    jlk_reg_write(&dev, 0x08, 0x0000);
    usleep(10000);

    /* 7. 尝试写JLK1263的REG(0x09) */
    jlk_reg_write(&dev, 0x09, 0x0001);
    usleep(10000);
    jlk_reg_read(&dev, 0x09, &val);
    printf("  After write 0x0001: REG(0x09)=0x%04X\n", val);
    jlk_reg_read(&dev, 0x0B, &val);
    printf("  REG(0x0B)=0x%04X (bit15=%d) %s\n", val, (val>>15)&1,
           (val & 0x8000) ? "" : "<<< CHANGED!");

    jlk_reg_write(&dev, 0x09, 0x0000);

    /* 8. 检查通道使能是否有效 */
    printf("\n=== Channel enable test ===\n");
    emif_set_ce(&dev, 0xb);
    jlk_reg_write(&dev, 0x03, 0xc3);
    usleep(50000);
    jlk_reg_read(&dev, 0x03, &val);
    printf("  After write 0xc3: REG(0x03)=0x%04X\n", val);
    jlk_reg_read(&dev, 0x0B, &val);
    printf("  REG(0x0B)=0x%04X (bit15=%d)\n", val, (val>>15)&1);

    xdma_close(&dev);
    return 0;
}
